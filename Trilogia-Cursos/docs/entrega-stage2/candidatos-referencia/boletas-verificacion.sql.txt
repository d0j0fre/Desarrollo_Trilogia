DECLARE @Authority sysname=N'$(AuthorityUser)',@App sysname=N'$(AppUser)';

 OR DB_NAME() IN(N'DistribuidoraJJ_DB',N'master',N'model',N'msdb',N'tempdb') THROW 59610,N'Private target ownership unaccredited.',1;

 OR @Authority LIKE N'%$(%' OR @App LIKE N'%$(%'
 OR NOT EXISTS(SELECT 1 FROM sys.database_principals WHERE name=@Authority AND type IN('S','U') AND principal_id>4)
 OR NOT EXISTS(SELECT 1 FROM sys.database_principals WHERE name=@App AND type IN('S','U') AND principal_id>4)
 THROW 59610,N'Restricted identities unaccredited.',1;

 THROW 59610,N'Candidate schema unavailable.',1;

 UNION SELECT OBJECT_ID(N'dbo.tr_PaySlipNotifications_BlockLegacyWrites');

-- The authority may inspect the three trigger parent tables, without SELECT.

 THROW 59610,N'Authority effective boundary differs.',1;

 THROW 59610,N'Public impersonation is not allowed.',1;

-- unrelated permissions are inventoried in the manifest rather than removed.
IF EXISTS(SELECT 1 FROM sys.database_permissions p WHERE p.grantee_principal_id=USER_ID(@Authority)

  (p.class=0 AND p.permission_name=N'CONNECT') OR
  (p.class=1 AND p.minor_id=0 AND p.permission_name=N'EXECUTE'
   AND p.major_id IN(OBJECT_ID(N'PaySlipNotifications.Request'),OBJECT_ID(N'PaySlipNotifications.Claim'),OBJECT_ID(N'PaySlipNotifications.Complete'),OBJECT_ID(N'PaySlipNotifications.Receipt'))) OR
  (p.class=1 AND p.minor_id=0 AND p.permission_name=N'VIEW DEFINITION'
   AND p.major_id IN(SELECT ObjectId FROM @Canonical UNION SELECT ObjectId FROM @CanonicalMetadataParents))))))
 THROW 59610,N'Authority has an unapproved direct grant.',1;
